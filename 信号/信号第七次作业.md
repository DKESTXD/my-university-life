A=10;
B=20;
N=20;
n=0:19;
x=A*cos(200*pi*n/1000)+B*sin(500*pi*n/1000);

subplot(2,1,1);
stem(0:N-1, abs(X));
title('幅度');
xlabel('k');
ylabel('幅度');

subplot(2,1,2);

title('相位');
